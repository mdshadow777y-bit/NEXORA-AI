async function sendMessage() { const message = input.value.trim(); if (!message || isBusy) { return; } stopListening(); stopSpeaking(); isBusy = true; sendButton.disabled = true; micButton.disabled = true; addMessageToUI("user", message); /* IMPORTANT: Current message is NOT included here. Worker adds the current message itself. */ const previousMessages = conversation .filter( item => item && ( item.role === "user" || item.role === "assistant" ) && typeof item.content === "string" ) .slice(-20); conversation.push({ role: "user", content: message }); saveConversation(); input.value = ""; showThinking(); try { const response = await fetch( API_URL, { method: "POST", headers: { "Content-Type": "application/json" }, body: JSON.stringify({ message: message, messages: previousMessages }) } ); /* IMPORTANT: Read response as TEXT first. This allows us to handle: 1. Normal JSON 2. SSE / streaming 3. Cloudflare/server error pages */ const rawText = await response.text(); console.log( "NEXORA HTTP STATUS:", response.status ); console.log( "NEXORA RAW RESPONSE:", rawText ); if (!response.ok) { let errorMessage = `HTTP ${response.status}`; try { const errorData = JSON.parse(rawText); errorMessage = errorData?.details?.error?.message || errorData?.message || errorData?.error || errorMessage; } catch { /* Server did not return JSON. Show a short version of the raw response. */ if (rawText) { errorMessage = rawText .replace(/<[^>]*>/g, " ") .replace(/\s+/g, " ") .trim() .slice(0, 500); } } throw new Error( errorMessage ); } let reply = ""; /* ===================================================== METHOD 1 — NORMAL JSON ===================================================== */ try { const data = JSON.parse(rawText); if ( typeof data?.reply === "string" ) { reply = data.reply.trim(); } /* Some APIs may return OpenAI-style format. */ if ( !reply && typeof data?.choices?.[0]?.message?.content === "string" ) { reply = data.choices[0] .message .content .trim(); } } catch { /* Not JSON. Continue and try SSE. */ } /* ===================================================== METHOD 2 — SSE / STREAMING ===================================================== */ if (!reply && rawText) { const lines = rawText.split(/\r?\n/); const chunks = []; for (const line of lines) { const trimmed = line.trim(); if ( !trimmed.startsWith("data:") ) { continue; } const dataPart = trimmed .slice(5) .trim(); if ( !dataPart || dataPart === "[DONE]" ) { continue; } try { const parsed = JSON.parse(dataPart); /* OpenAI / OpenRouter style */ const content = parsed?.choices?.[0]?.delta?.content || parsed?.choices?.[0]?.message?.content || parsed?.reply || ""; if (content) { chunks.push(content); } } catch { /* Plain SSE text */ chunks.push( dataPart ); } } reply = chunks.join("").trim(); } /* ===================================================== NOTHING FOUND ===================================================== */ if (!reply) { console.error( "NEXORA could not parse server response:", rawText ); throw new Error( "Server response could not be understood. Open browser Console (F12) and check 'NEXORA RAW RESPONSE'." ); } /* ===================================================== SUCCESS ===================================================== */ removeThinking(); addMessageToUI( "assistant", reply ); conversation.push({ role: "assistant", content: reply }); saveConversation(); if (voiceEnabled) { speak(reply); } } catch (error) { console.error( "NEXORA REQUEST ERROR:", error ); removeThinking(); addMessageToUI( "assistant", `⚠️ NEXORA request failed.\n\n${error.message}` ); } isBusy = false; sendButton.disabled = false; micButton.disabled = false; input.focus(); }
